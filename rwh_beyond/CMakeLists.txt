cmake_minimum_required(VERSION 3.16)
project(rwh_beyond)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(nav2_util REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(rwh_beyond_msgs REQUIRED)
set(dependencies rclcpp rclcpp_lifecycle rclcpp_components nav2_util lifecycle_msgs rwh_beyond_msgs)
add_library(${PROJECT_NAME}_core SHARED src/beyond_server.cpp)
target_compile_features(${PROJECT_NAME}_core PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME}_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME}_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}_core ${dependencies})
rclcpp_components_register_nodes(${PROJECT_NAME}_core "rwh_beyond::BeyondServer")
add_executable(rwh_beyond_server src/main.cpp)
target_link_libraries(rwh_beyond_server ${PROJECT_NAME}_core)
ament_target_dependencies(rwh_beyond_server ${dependencies})
install(TARGETS ${PROJECT_NAME}_core EXPORT ${PROJECT_NAME}Targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS rwh_beyond_server RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(${PROJECT_NAME}Targets HAS_LIBRARY_TARGET)
ament_export_dependencies(${dependencies})
ament_package()
